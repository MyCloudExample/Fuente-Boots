#include "pico/stdlib.h"
#include "hardware/pwm.h"
#include "hardware/adc.h"
#include "hardware/i2c.h"
#include "lcd.h"  // ← Usa TU librería existente

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#include "FreeRTOS.h"
#include "task.h"
#include "queue.h"

// ================= CONFIGURACIÓN =================
#define PWM_PIN     22
#define POT_PIN     26
#define VOUT_PIN    27
#define LED_PIN     25
#define PWM_FREQ    30000

// LCD I2C (ya configurado en tu lcd.h)
#define I2C_PORT    i2c0
#define I2C_SDA     4
#define I2C_SCL     5
#define LCD_ADDR    0x27

// Límites
#define DUTY_MIN    58.0f
#define DUTY_MAX    80.0f

// Divisor
#define R1          81700.0f
#define R2          9960.0f
#define DIV_RATIO   (R2 / (R1 + R2))

// ================= VARIABLES =================
static uint32_t pwm_wrap;
static uint slice_num, channel;
QueueHandle_t duty_queue;

// ================= FUNCIONES ADC =================
float read_potentiometer() {
    adc_select_input(0);
    uint16_t raw = adc_read();
    
    float duty = DUTY_MIN + ((float)raw / 4095.0f) * (DUTY_MAX - DUTY_MIN);
    
    if (duty < DUTY_MIN) duty = DUTY_MIN;
    if (duty > DUTY_MAX) duty = DUTY_MAX;
    
    return duty;
}

float read_output_voltage() {
    adc_select_input(1);
    uint16_t raw = adc_read();
    
    float v_adc = (raw / 4095.0f) * 3.3f;
    float v_out = v_adc / DIV_RATIO;
    
    return v_out;
}

// ================= TAREAS FREERTOS =================
void task_potentiometer(void *pv) {
    adc_init();
    adc_gpio_init(POT_PIN);
    adc_gpio_init(VOUT_PIN);
    
    float last_duty = DUTY_MIN;
    
    while (1) {
        float duty = read_potentiometer();
        
        if (fabs(duty - last_duty) > 0.1f) {
            xQueueSend(duty_queue, &duty, 0);
            last_duty = duty;
        }
        
        vTaskDelay(pdMS_TO_TICKS(20));
    }
}

void task_pwm_control(void *pv) {
    float duty = DUTY_MIN;
    
    while (1) {
        if (xQueueReceive(duty_queue, &duty, portMAX_DELAY)) {
            uint32_t level = (uint32_t)((duty / 100.0f) * pwm_wrap);
            pwm_set_chan_level(slice_num, channel, level);
        }
    }
}

void task_lcd_display(void *pv) {
    char buffer[21];
    float duty = DUTY_MIN;
    float voltage = 0;
    
    // Inicializar LCD usando TU librería
    lcd_init();
    lcd_clear();
    
    // Mensaje inicial
    lcd_set_cursor(0, 0);
    lcd_string("Fuente Boost 30kHz");
    lcd_set_cursor(1, 0);
    lcd_string("Duty: 58-80%");
    lcd_set_cursor(2, 0);
    lcd_string("Vout: 12-24V");
    lcd_set_cursor(3, 0);
    lcd_string("Iniciando...");
    
    vTaskDelay(pdMS_TO_TICKS(2000));
    
    while (1) {
        // Leer el último duty cycle
        xQueuePeek(duty_queue, &duty, 0);
        
        // Leer voltaje de salida
        voltage = read_output_voltage();
        
        // Actualizar LCD - línea por línea
        lcd_set_cursor(0, 0);
        snprintf(buffer, 21, "Duty: %5.1f%%        ", duty);
        lcd_string(buffer);
        
        lcd_set_cursor(1, 0);
        snprintf(buffer, 21, "Vout: %5.2f V        ", voltage);
        lcd_string(buffer);
        
        lcd_set_cursor(2, 0);
        snprintf(buffer, 21, "ADC: %4d            ", adc_read());
        lcd_string(buffer);
        
        lcd_set_cursor(3, 0);
        snprintf(buffer, 21, "PWM: %4u/%4u        ", 
                (uint32_t)((duty/100.0f)*pwm_wrap), pwm_wrap);
        lcd_string(buffer);
        
        // LED parpadea
        gpio_xor_mask(1 << LED_PIN);
        
        vTaskDelay(pdMS_TO_TICKS(500));
    }
}

// ================= MAIN =================
int main() {
    // Inicializar I2C (ya lo hace lcd_init)
    i2c_init(I2C_PORT, 100000);
    gpio_set_function(I2C_SDA, GPIO_FUNC_I2C);
    gpio_set_function(I2C_SCL, GPIO_FUNC_I2C);
    gpio_pull_up(I2C_SDA);
    gpio_pull_up(I2C_SCL);
    
    // LED
    gpio_init(LED_PIN);
    gpio_set_dir(LED_PIN, GPIO_OUT);
    gpio_put(LED_PIN, 1);
    
    // PWM
    gpio_set_function(PWM_PIN, GPIO_FUNC_PWM);
    slice_num = pwm_gpio_to_slice_num(PWM_PIN);
    channel = pwm_gpio_to_channel(PWM_PIN);
    
    // Calcular para 30 kHz
    uint32_t clock = 125000000;
    pwm_wrap = (clock / PWM_FREQ) - 1;
    
    pwm_config config = pwm_get_default_config();
    pwm_config_set_clkdiv(&config, 1.0f);
    pwm_config_set_wrap(&config, pwm_wrap);
    pwm_init(slice_num, &config, true);
    
    // Iniciar con duty mínimo
    uint32_t init_level = (uint32_t)((DUTY_MIN / 100.0f) * pwm_wrap);
    pwm_set_chan_level(slice_num, channel, init_level);
    
    // Crear cola
    duty_queue = xQueueCreate(5, sizeof(float));
    
    // Crear tareas
    xTaskCreate(task_potentiometer, "Pot", 256, NULL, 2, NULL);
    xTaskCreate(task_pwm_control, "PWM", 256, NULL, 3, NULL);
    xTaskCreate(task_lcd_display, "LCD", 512, NULL, 1, NULL);
    
    // Iniciar scheduler
    vTaskStartScheduler();
    
    while (1) {
        gpio_xor_mask(1 << LED_PIN);
        sleep_ms(500);
    }
}
